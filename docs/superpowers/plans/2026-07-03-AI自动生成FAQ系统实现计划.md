> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 创建 FAQ 应用 + 热点问题接口，实现 AI 自动提取问答日志 → 聚类 → LLM 生成 → 审核发布

**Architecture:** 新建 `apps/faq` 应用（模型/序列化器/视图/管理员/Celery 任务），在 `apps/chat` 追加热点问题接口。DeepSeek Embedding 做向量聚类，DeepSeek Chat 做草稿生成。Celery Beat 每周一凌晨驱动全流程。

**Tech Stack:** Django 4.2 + DRF + Celery/Redis + DeepSeek API

## Global Constraints

- 使用 Django 4.2 + MySQL 8.0
- 用户模型 `apps.accounts.User`，role 字段含 `super_admin` / `college_admin` / `dept_admin` / `admin` / `user`
- DeepSeek API 通过 `utils.llm.llm_client` 全局单例调用
- Celery 已配置 `app.autodiscover_tasks()`，新任务放 `apps/faq/tasks.py` 即可自动发现
- 所有 API 需要 `rest_framework.permissions.IsAuthenticated` + 自定义权限类
- 权限类位于 `config/permissions.py`
- 代码风格匹配现有项目：中文注释 + type hints + logger
- `FAQItem.college` = NULL 表示全局 FAQ；非 NULL 表示学院专属

from django.conf import settings
from django.db import models


class FAQCategory(models.Model):
    """FAQ 分类"""
    name = models.CharField(max_length=100, verbose_name='分类名称')
    icon = models.CharField(max_length=50, blank=True, default='', verbose_name='图标')
    sort_order = models.IntegerField(default=0, verbose_name='排序')
    created_at = models.DateTimeField(auto_now_add=True, verbose_name='创建时间')

    class Meta:
        db_table = 'faq_category'
        verbose_name = 'FAQ 分类'
        verbose_name_plural = verbose_name
        ordering = ['sort_order', 'id']

    def __str__(self):
        return self.name


class FAQItem(models.Model):
    """FAQ 条目"""
    STATUS_CHOICES = [
        ('draft', '草稿'),
        ('published', '已发布'),
        ('rejected', '已驳回'),
    ]

    question = models.CharField(max_length=500, verbose_name='标准化问题')
    answer = models.TextField(verbose_name='标准答案')
    category = models.ForeignKey(
        FAQCategory,
        on_delete=models.SET_NULL,
        null=True, blank=True,
        related_name='faq_items',
        verbose_name='所属分类',
    )
    college = models.ForeignKey(
        'accounts.College',
        on_delete=models.SET_NULL,
        null=True, blank=True,
        related_name='faq_items',
        verbose_name='所属学院（NULL=全局）',
    )
    tags = models.JSONField(default=list, blank=True, verbose_name='关键词标签')
    status = models.CharField(
        max_length=20, choices=STATUS_CHOICES, default='draft',
        db_index=True, verbose_name='状态',
    )
    frequency = models.FloatField(default=0.0, verbose_name='该问题占比(%)')
    variant_count = models.IntegerField(default=0, verbose_name='相似变体合并数')
    source_period_start = models.DateTimeField(null=True, blank=True, verbose_name='分析周期起始')
    source_period_end = models.DateTimeField(null=True, blank=True, verbose_name='分析周期结束')
    related_doc_ids = models.JSONField(default=list, blank=True, verbose_name='关联知识库文档ID')
    reviewed_by = models.ForeignKey(
        settings.AUTH_USER_MODEL,
        on_delete=models.SET_NULL,
        null=True, blank=True,
        related_name='reviewed_faqs',
        verbose_name='审核人',
    )
    reviewed_at = models.DateTimeField(null=True, blank=True, verbose_name='审核时间')
    created_at = models.DateTimeField(auto_now_add=True, verbose_name='创建时间')
    updated_at = models.DateTimeField(auto_now=True, verbose_name='更新时间')

    class Meta:
        db_table = 'faq_item'
        verbose_name = 'FAQ 条目'
        verbose_name_plural = verbose_name
        ordering = ['-frequency', '-created_at']
        indexes = [

@admin.register(FAQCategory)
class FAQCategoryAdmin(admin.ModelAdmin):
    list_display = ['id', 'name', 'sort_order', 'created_at']
    search_fields = ['name']
    list_editable = ['sort_order']


@admin.register(FAQItem)
class FAQItemAdmin(admin.ModelAdmin):
    list_display = [
        'id', 'question_short', 'category', 'college', 'frequency',
        'variant_count', 'status', 'reviewed_by', 'created_at',
    ]
    list_filter = ['status', 'category', 'college', 'created_at']
    search_fields = ['question', 'answer', 'tags']
    readonly_fields = [
        'frequency', 'variant_count', 'source_period_start', 'source_period_end',
        'related_doc_ids', 'reviewed_by', 'reviewed_at', 'created_at', 'updated_at',
    ]
    actions = ['publish_selected', 'reject_selected']

    def question_short(self, obj):
        return obj.question[:40]
    question_short.short_description = '问题'

    def publish_selected(self, request, queryset):
        from django.utils import timezone
        updated = queryset.filter(status='draft').update(
            status='published',
            reviewed_by=request.user,
            reviewed_at=timezone.now(),
        )
        self.message_user(request, f'已发布 {updated} 条 FAQ')
    publish_selected.short_description = '发布选中草稿'

    def reject_selected(self, request, queryset):
        from django.utils import timezone
        updated = queryset.filter(status='draft').update(
            status='rejected',
            reviewed_by=request.user,
            reviewed_at=timezone.now(),
        )
        self.message_user(request, f'已驳回 {updated} 条 FAQ')
    reject_selected.short_description = '驳回选中草稿'


@admin.register(FAQGenerationLog)
class FAQGenerationLogAdmin(admin.ModelAdmin):
    list_display = [
        'id', 'status', 'total_questions', 'clusters_found',

class IsCollegeAdmin(permissions.BasePermission):
    """
    学院管理员权限 — super_admin 或 college_admin 可访问。
    学院管理员的操作范围限定在本学院内。
    """

    def has_permission(self, request, view):
        return (
            request.user
            and request.user.is_authenticated
            and request.user.role in ('super_admin', 'college_admin')
        )


class IsFAQAdmin(permissions.BasePermission):
    """
    FAQ 管理员权限 — super_admin / college_admin / dept_admin 可访问。
    不同角色有不同数据范围：
    - super_admin: 全部
    - college_admin: 所属学院的 FAQ
    - dept_admin: 所属部门的学院范围内 FAQ
    """

    def has_permission(self, request, view):
        return (
            request.user
            and request.user.is_authenticated
            and request.user.role in ('super_admin', 'college_admin', 'dept_admin', 'admin')
        )

from django.utils import timezone
from rest_framework import generics, permissions, status
from rest_framework.response import Response

from config.permissions import IsFAQAdmin, IsSuperAdmin

from .models import FAQCategory, FAQItem
from .serializers import (

    FAQItemDetailSerializer,

logger = logging.getLogger(__name__)

class FAQCategoryListView(generics.ListAPIView):
    """
    GET /api/faq/categories/
    获取 FAQ 分类列表
    """
    queryset = FAQCategory.objects.all()
    serializer_class = FAQCategorySerializer
    permission_classes = [permissions.IsAuthenticated]
    pagination_class = None


class FAQItemListView(generics.ListAPIView):
    """
    GET /api/faq/items/
    获取已发布的 FAQ 列表

    规则：
    - 仅返回 status='published' 的记录
    - 当前用户 college_id 匹配 college（NULL 的全局 FAQ 也可见）
    - 按 frequency 降序排列
    """
    serializer_class = FAQItemListSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        user = self.request.user
        return FAQItem.objects.filter(
            status='published',
        ).filter(
            college__isnull=True
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        ).order_by('-frequency', '-created_at')


class FAQItemDetailView(generics.RetrieveAPIView):
    """
    GET /api/faq/items/{id}/
    FAQ 详情
    """
    serializer_class = FAQItemDetailSerializer
    permission_classes = [permissions.IsAuthenticated]

    def get_queryset(self):
        user = self.request.user
        return FAQItem.objects.filter(
            status='published',
            college__isnull=True
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        )


# ======================================================================
# 管理 API（供管理员后台调用）
# ======================================================================

class FAQDraftListView(generics.ListAPIView):
    """
    GET /api/faq/drafts/
    待审核草稿列表

    权限：
    - super_admin: 全部
    - college_admin: 本学院 + 全局
    """
    serializer_class = FAQItemListSerializer
    permission_classes = [permissions.IsAuthenticated, IsFAQAdmin]

    def get_queryset(self):
        user = self.request.user
        qs = FAQItem.objects.filter(status='draft')

        if user.role == 'super_admin':
            return qs.order_by('-frequency', '-created_at')

        # college_admin / dept_admin / admin: 只看本学院的和全局的
        return qs.filter(
            models.Q(college__isnull=True)
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        ).order_by('-frequency', '-created_at')


class FAQDraftDetailView(generics.RetrieveUpdateAPIView):
    """
    GET/PATCH /api/faq/drafts/{id}/
    草稿详情 / 编辑草稿
    """
    serializer_class = FAQItemDraftSerializer
    permission_classes = [permissions.IsAuthenticated, IsFAQAdmin]

    def get_queryset(self):
        user = self.request.user
        qs = FAQItem.objects.filter(status='draft')
        if user.role == 'super_admin':
            return qs
        return qs.filter(
            models.Q(college__isnull=True)
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        )

    def perform_update(self, serializer):
        serializer.save()


class FAQDraftActionView(generics.GenericAPIView):
    """
    POST /api/faq/drafts/{id}/publish/  — 发布
    POST /api/faq/drafts/{id}/reject/  — 驳回
    """
    permission_classes = [permissions.IsAuthenticated, IsFAQAdmin]
    serializer_class = FAQItemPublishSerializer

    def get_queryset(self):
        user = self.request.user
        qs = FAQItem.objects.filter(status='draft')
        if user.role == 'super_admin':
            return qs
        return qs.filter(
            models.Q(college__isnull=True)
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        )

    def post(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        faq = self.get_object()
        action = serializer.validated_data['action']

        if action == 'publish':
            faq.status = 'published'
        else:
            faq.status = 'rejected'

        faq.reviewed_by = request.user
        faq.reviewed_at = timezone.now()
        faq.save(update_fields=['status', 'reviewed_by', 'reviewed_at', 'updated_at'])

        logger.info(
            'FAQ %s: id=%d by user=%s', action, faq.id, request.user.username,
        )

        return Response(
            {'status': 'ok', 'faq_id': faq.id, 'new_status': faq.status},
        )


class FAQManageListView(generics.ListAPIView):
    """
    GET /api/faq/items/  (管理端)
    全部 FAQ 列表（含草稿），管理员用
    """
    serializer_class = FAQItemListSerializer
    permission_classes = [permissions.IsAuthenticated, IsFAQAdmin]

    def get_queryset(self):
        user = self.request.user
        if user.role == 'super_admin':
            return FAQItem.objects.all().order_by('-frequency', '-created_at')

        return FAQItem.objects.filter(
            models.Q(college__isnull=True)
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        ).order_by('-frequency', '-created_at')


class FAQManageDeleteView(generics.DestroyAPIView):
    """
    DELETE /api/faq/items/{id}/
    删除 FAQ
    """
    permission_classes = [permissions.IsAuthenticated, IsFAQAdmin]

    def get_queryset(self):
        user = self.request.user
        if user.role == 'super_admin':
            return FAQItem.objects.all()

            models.Q(college__isnull=True)
            | (models.Q(college_id=user.college_id) if user.college_id else models.Q())
        )

class HotQuestionsView(APIView):
    """
    热点问题广场 — "大家都在问"

    GET /api/chat/hot-questions/?days=7&top_k=10

    实时统计近期用户高频提问，按提问次数降序排列。
    适合展示在智能问答主界面。
    """
    permission_classes = [permissions.IsAuthenticated]

    def get(self, request, *args, **kwargs):
        days = int(request.query_params.get('days', 7))
        top_k = int(request.query_params.get('top_k', 10))

        since = timezone.now() - timezone.timedelta(days=days)
        cutoff = timezone.make_aware(
            timezone.datetime.combine(since.date(), timezone.datetime.min.time())
        ) if timezone.is_naive(since) else since

        # 聚合统计：按问题文本分组，统计出现次数
        from .models import ChatMessage

        questions_qs = (
            ChatMessage.objects
            .filter(role='user', created_at__gte=cutoff)
            .values('content')
            .annotate(count=Count('id'))
            .order_by('-count')[:top_k]
        )

        results = [
            {
                'question': item['content'],
                'count': item['count'],
            }
            for item in questions_qs
        ]

        return Response(results)
```

```python
"""
FAQ 提取管道辅助函数

依赖:
- utils.llm.llm_client (DeepSeek API 封装)
- numpy (向量运算)
"""
import logging
from typing import List, Tuple, Optional

import numpy as np
from django.conf import settings
from utils.llm import llm_client

logger = logging.getLogger(__name__)


class QuestionCluster:
    """问题簇 — 由相似问题聚合而成"""
    def __init__(self, questions: List[str], answers: List[str], vector: List[float]):
        self.questions = questions      # 所有原始问题变体
        self.answers = answers          # 对应的 AI 回答
        self.vector = vector            # 簇心向量（第一个问题的向量）
        self.size = len(questions)      # 簇大小

    def to_dict(self) -> dict:
        return {
            'questions': self.questions,
            'answers': self.answers,
            'size': self.size,
        }


def cosine_similarity(v1: List[float], v2: List[float]) -> float:
    """计算两个向量的余弦相似度"""
    a = np.array(v1, dtype=np.float32)
    b = np.array(v2, dtype=np.float32)
    norm_a = np.linalg.norm(a)
    norm_b = np.linalg.norm(b)
    if norm_a == 0 or norm_b == 0:
        return 0.0
    return float(np.dot(a, b) / (norm_a * norm_b))


def embed_questions(questions: List[str]) -> List[List[float]]:
    """
    批量获取问题向量。

    Args:
        questions: 问题文本列表

    Returns:
        对应的向量列表
    """
    if not questions:
        return []

    vectors = llm_client.embed(questions)
    logger.debug(f'Embedding {len(questions)} 个问题完成')
    return vectors


def cluster_similar_questions(
    question_answer_pairs: List[Tuple[str, str]],
    similarity_threshold: float = 0.85,
) -> List[QuestionCluster]:
    """
    将相似问题聚类（阶段2：向量聚类）。

    Args:
        question_answer_pairs: [(问题, 对应AI回答), ...]
        similarity_threshold: 相似度阈值，默认 0.85

    Returns:
        问题簇列表
    """
    if not question_answer_pairs:
        return []

    questions = [pair[0] for pair in question_answer_pairs]
    vectors = embed_questions(questions)

    if not vectors or len(vectors) != len(questions):
        logger.error('Embedding 失败，返回空簇')
        return []

    # 贪心聚类：依次处理每个问题
    clusters: List[QuestionCluster] = []
    assigned = [False] * len(questions)

    for i in range(len(questions)):
        if assigned[i]:
            continue

        cluster_questions = [questions[i]]
        cluster_answers = [question_answer_pairs[i][1]]
        assigned[i] = True

        for j in range(i + 1, len(questions)):
            if assigned[j]:
                continue

            sim = cosine_similarity(vectors[i], vectors[j])
            if sim >= similarity_threshold:
                cluster_questions.append(questions[j])
                cluster_answers.append(question_answer_pairs[j][1])
                assigned[j] = True

        clusters.append(QuestionCluster(
            questions=cluster_questions,
            answers=cluster_answers,
            vector=vectors[i],
        ))

    logger.info(f'聚类完成: {len(question_answer_pairs)} 个问题 → {len(clusters)} 个簇')
    for idx, c in enumerate(clusters):
        logger.debug(f'  簇 {idx+1}: size={c.size}, 代表="{c.questions[0][:40]}..."')

    return clusters


def filter_by_ratio(
    clusters: List[QuestionCluster],
    total_count: int,
    min_ratio: float = None,
    max_count: int = None,
) -> List[QuestionCluster]:
    """
    按比例筛选高频簇（阶段3）。

    Args:
        clusters: 问题簇列表
        total_count: 总提问数
        min_ratio: 最小占比阈值（百分比），默认 settings.FAQ_MIN_RATIO
        max_count: 最大保留数，默认 settings.FAQ_MAX_CLUSTERS

    Returns:
        筛选后的高频问题簇（按占比降序）
    """
    min_ratio = min_ratio or getattr(settings, 'FAQ_MIN_RATIO', 0.5)
    max_count = max_count or getattr(settings, 'FAQ_MAX_CLUSTERS', 20)

    if total_count <= 0:
        return []

    # 计算占比
    scored = []
    for cluster in clusters:
        ratio = (cluster.size / total_count) * 100.0
        scored.append((cluster, ratio))

    # 过滤 + 排序
    filtered = [(c, r) for c, r in scored if r >= min_ratio]
    filtered.sort(key=lambda x: x[1], reverse=True)

    # 取 Top-K
    top = filtered[:max_count]

    logger.info(
        f'比例筛选: {len(clusters)} 簇 → {len(top)} 簇通过 '
        f'(threshold={min_ratio}%, top={max_count})'
    )

    return [c for c, _ in top]


def generate_faq_draft(cluster: QuestionCluster) -> Optional[dict]:
    """
    对单个高频问题簇调用 LLM 生成 FAQ 草稿（阶段4）。

    Args:
        cluster: 问题簇

    Returns:
        {
            'question': str,           # 标准化问题
            'answer': str,             # 标准答案
            'tags': List[str],         # 关键词标签
            'college_id': int | None,  # 推断的学院ID，None=全局
        }
        失败返回 None
    """
    # 构建 LLM prompt
    questions_text = '\n'.join(f'- {q}' for q in cluster.questions[:10])
    answers_text = '\n'.join(f'- {a[:200]}' for a in cluster.answers[:5] if a)

    system_prompt = """你是一个 FAQ 生成助手。请根据用户反复询问的同类问题变体和已有回答，
生成一份标准化的 FAQ 条目。

要求：
1. 将多个问题变体归纳为一个清晰、简洁的标准化问题
2. 综合已有回答，生成完整的标准答案（200-500字，条理清晰）
3. 提取 3-5 个关键词标签
4. 判断是否与特定学院相关：
   - 如果问题明确提到某个学院名称，输出该学院名称
   - 如果是通用问题（如"如何重置密码"），输出"全局"
5. 以 JSON 格式输出，不要包含其他内容"""

    user_prompt = f"""问题变体（共 {cluster.size} 条，以下展示前 10 条）：
{questions_text}

已有回答（前 5 条摘要）：
{answers_text}

请输出如下 JSON（不要有其他内容）：
{{
    "question": "标准化后的问题",
    "answer": "完整的标准答案",
    "tags": ["标签1", "标签2", "标签3"],
    "college_name": "学院名称 或 '全局'"
}}"""

    try:
        import json as _json

        response = llm_client.chat(
            messages=[
                {'role': 'system', 'content': system_prompt},
                {'role': 'user', 'content': user_prompt},
            ],
            temperature=0.3,
            max_tokens=2048,
        )

        if not response:
            logger.error(f'LLM 生成 FAQ 草稿失败：返回空')
            return None

        # 提取 JSON（LLM 可能用 ```json 包裹）
        text = response.strip()
        if '```json' in text:
            text = text.split('```json')[1].split('```')[0].strip()
        elif '```' in text:
            text = text.split('```')[1].split('```')[0].strip()

        result = _json.loads(text)

        # 解析 college_id
        college_id = None
        college_name = result.get('college_name', '全局')
        if college_name and college_name != '全局':
            try:
                from apps.accounts.models import College
                college = College.objects.filter(name__icontains=college_name).first()
                if college:
                    college_id = college.id
            except Exception as e:
                logger.warning(f'解析学院名称失败: {e}')

        return {
            'question': result.get('question', cluster.questions[0][:500]),
            'answer': result.get('answer', ''),
            'tags': result.get('tags', []),
            'college_id': college_id,
        }

    except Exception as e:
        logger.error(f'LLM 生成 FAQ 草稿失败: {e}')
        return None
```

```python
"""
FAQ 自动生成 — Celery 定时任务

流程:
  阶段1: 采集近一周用户问题
  阶段2: 向量聚类合并相似问题
  阶段3: 按占比筛选高频簇
  阶段4: LLM 生成 FAQ 草稿

调度: 每周一 03:00（通过 Celery Beat）
"""
import logging
from datetime import timedelta

from celery import shared_task
from django.utils import timezone

from apps.chat.models import ChatMessage
from utils.llm import llm_client

from .models import FAQItem, FAQGenerationLog
from .utils import (
    QuestionCluster,
    cluster_similar_questions,
    filter_by_ratio,
    generate_faq_draft,
)

logger = logging.getLogger(__name__)


@shared_task(bind=True, max_retries=2, default_retry_delay=120)
def generate_faq_weekly(self):
    """
    主入口：每周 FAQ 自动提取流程。

    由 Celery Beat 定时触发（每周一 03:00）。
    也可手动调用: python manage.py shell -c "from apps.faq.tasks import generate_faq_weekly; generate_faq_weekly()"
    """
    # 防重叠：检查是否有正在运行的
    running_log = FAQGenerationLog.objects.filter(status='running').first()
    if running_log:
        # 超过 2 小时视为僵死，允许重新运行
        if running_log.started_at and running_log.started_at > timezone.now() - timedelta(hours=2):
            logger.warning('FAQ 生成任务已在运行中（log_id=%d），跳过本次触发', running_log.id)
            return
        else:
            logger.warning('发现僵死的 FAQ 生成任务（log_id=%d），重新开始', running_log.id)
            running_log.status = 'failed'
            running_log.error_message = '超时自动标记为失败'
            running_log.completed_at = timezone.now()
            running_log.save(update_fields=['status', 'error_message', 'completed_at'])

    # 创建运行日志
    log = FAQGenerationLog.objects.create(status='running')
    logger.info('FAQ 自动生成开始（log_id=%d）', log.id)

    try:
        # ---- 阶段1: 采集 ----
        since = timezone.now() - timedelta(days=7)
        messages = (
            ChatMessage.objects
            .filter(role='user', created_at__gte=since)
            .select_related('conversation')
            .only('content', 'conversation_id', 'created_at')
        )

        total_questions = messages.count()
        logger.info('阶段1: 采集近7天用户问题 %d 条', total_questions)

        if total_questions == 0:
            log.status = 'completed'
            log.total_questions = 0
            log.completed_at = timezone.now()
            log.save(update_fields=['status', 'total_questions', 'completed_at'])
            logger.info('无用户问题日志，FAQ 生成结束')
            return

        # 去重 + 获取对应 AI 回答
        seen = set()
        question_answer_pairs = []
        for msg in messages:
            content = msg.content.strip()
            if not content or content in seen:
                continue
            seen.add(content)

            # 获取对应 AI 回答（紧跟着的同会话下一条 assistant 消息）
            ai_reply = (
                ChatMessage.objects
                .filter(
                    conversation_id=msg.conversation_id,
                    role='assistant',
                    created_at__gt=msg.created_at,
                )
                .order_by('created_at')
                .first()
            )
            answer = ai_reply.content if ai_reply else ''

            question_answer_pairs.append((content, answer))

        log.total_questions = total_questions
        log.save(update_fields=['total_questions'])
        logger.info('去重后获得 %d 个独立问题', len(question_answer_pairs))

        # ---- 阶段2: 向量聚类 ----
        clusters = cluster_similar_questions(question_answer_pairs, similarity_threshold=0.85)
        log.clusters_found = len(clusters)
        log.save(update_fields=['clusters_found'])

        if not clusters:
            log.status = 'completed'
            log.completed_at = timezone.now()
            log.save(update_fields=['status', 'completed_at'])
            logger.info('未形成有效问题簇，FAQ 生成结束')
            return

        # ---- 阶段3: 比例筛选 ----
        high_freq_clusters = filter_by_ratio(clusters, total_questions)
        logger.info('阶段3: 高频簇 %d 个', len(high_freq_clusters))

        if not high_freq_clusters:
            log.status = 'completed'
            log.completed_at = timezone.now()
            log.save(update_fields=['status', 'completed_at'])
            logger.info('无高频问题簇，FAQ 生成结束')
            return

        # ---- 阶段4: 生成 FAQ 草稿 ----
        drafts_created = 0
        for cluster in high_freq_clusters:
            try:
                # 跳过与已有 FAQ 过度相似的问题
                representative = cluster.questions[0][:200]
                existing = FAQItem.objects.filter(
                    status__in=['draft', 'published'],
                ).order_by('-created_at')[:50]

                # 向量相似度比较（取第一个问题向量做快速检查）
                cluster_vector = llm_client.embed_one(representative)
                should_skip = False
                for exist in existing:
                    if not exist.question:
                        continue
                    exist_vector = llm_client.embed_one(exist.question[:200])
                    from .utils import cosine_similarity
                    if cosine_similarity(cluster_vector, exist_vector) >= 0.90:
                        should_skip = True
                        logger.info('跳过已存在的 FAQ: %s（相似度 >= 0.90）', representative[:40])
                        break

                if should_skip:
                    continue

                # 调用 LLM 生成
                result = generate_faq_draft(cluster)
                if not result:
                    logger.warning('LLM 生成失败，跳过该簇')
                    continue

                # 保存为草稿
                FAQItem.objects.create(
                    question=result['question'][:500],
                    answer=result['answer'],
                    tags=result.get('tags', []),
                    college_id=result.get('college_id'),
                    status='draft',
                    frequency=round((cluster.size / total_questions) * 100, 2),
                    variant_count=cluster.size,
                    source_period_start=since,
                    source_period_end=timezone.now(),
                )
                drafts_created += 1
                logger.info('生成 FAQ 草稿: %s', result['question'][:40])

            except Exception as e:
                logger.error(f'处理问题簇时出错: {e}')
                continue

        # 完成
        log.status = 'completed'
        log.drafts_generated = drafts_created
        log.completed_at = timezone.now()
        log.save(update_fields=['status', 'drafts_generated', 'completed_at'])
        logger.info('FAQ 自动生成完成: 生成 %d 条草稿', drafts_created)

    except Exception as e:
        logger.error(f'FAQ 自动生成失败: {e}')
        log.status = 'failed'
        log.error_message = str(e)[:2000]
        log.completed_at = timezone.now()
        log.save(update_fields=['status', 'error_message', 'completed_at'])
        raise self.retry(exc=e)
```
